Leitura automática das etiquetas

Controle de Oxigênio Medicinal — AME

As etiquetas podem estar em qualquer posição. Faça uma captura da etiqueta do fornecedor/produto e depois uma captura da identificação individual do cilindro. O sistema combina as duas leituras automaticamente.

2 — Etiqueta individual do cilindro

Aguardando a primeira etiqueta.

O número de série do fabricante permanece separado. O código de barras só será preenchido quando o leitor conseguir decodificá-lo; ele nunca será substituído pelo lote.

// ========================================== // INICIAR CÂMERA // ========================================== async function start() { try { if (!target) { throw new Error( 'Abra esta página pelo botão do recebimento.' ); } if (!navigator.mediaDevices?.getUserMedia) { throw new Error( 'Este navegador não disponibiliza a câmera.' ); } stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: { ideal: 'environment' }, width: { ideal: 1920 }, height: { ideal: 1080 } }, audio:false }); $('video').srcObject = stream; await $('video').play(); // Leitor de código de barras, quando disponível. if ('BarcodeDetector' in window) { try { detector = new BarcodeDetector({ formats:[ 'code_128', 'code_39', 'ean_13', 'ean_8', 'itf', 'qr_code', 'codabar', 'upc_a', 'upc_e' ] }); } catch(e) { detector = null; } } say( 'Câmera ativa. Primeiro fotografe a etiqueta do fornecedor/produto.', 'ok' ); } catch(e) { say( 'Não foi possível iniciar a câmera: ' + e.message, 'error' ); } } // ========================================== // CAPTURAR E PROCESSAR ETIQUETAS // ========================================== async function capture(kind) { if (working || !stream) return; const video = $('video'); if (!video.videoWidth) { say( 'Aguarde a câmera iniciar.', 'error' ); return; } working = true; const canvas = document.createElement('canvas'); canvas.width = video.videoWidth; canvas.height = video.videoHeight; canvas.getContext('2d').drawImage( video, 0, 0, canvas.width, canvas.height ); say( kind === 'product' ? 'Lendo etiqueta do produto, lote e data...' : 'Lendo identificação individual do cilindro...' ); try { if (!worker) { worker = await Tesseract.createWorker( 'eng', 1, { logger:m => { if (m.status === 'recognizing text') { say( 'Reconhecendo texto: ' + Math.round((m.progress || 0) * 100) + '%' ); } } } ); } // Ajusta o OCR para o tipo de etiqueta fotografada. // Produto: bloco com código, lote e data. // Cilindro: uma linha numérica, evitando confundir com outros textos. if (worker.setParameters) { await worker.setParameters( kind === 'product' ? { tessedit_pageseg_mode: '6', tessedit_char_whitelist: 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789/:-.', preserve_interword_spaces: '1' } : { tessedit_pageseg_mode: '7', tessedit_char_whitelist: '0123456789', preserve_interword_spaces: '1' } ); } const result = await worker.recognize(canvas);
